\item\label{p:rank}
Consider the matrix $\vec A$ below:
\begin{align}
\vec A=\myvec{2&3&4&5\\0&6&7&8\\0&0&\alpha&\beta\\0&0&0&\gamma}
\label{eq:P1-A}
\end{align}
For which of the following combinations of $\alpha$, $\beta$ and $\gamma$ is the rank of $\vec A$ at least three?
\hfill (EC~2025)
\begin{conds}
\item $\alpha=0$ and $\beta=\gamma\ne0$.
\item $\alpha=\beta=\gamma=0$.
\item $\beta=\gamma=0$ and $\alpha\ne0$.
\item $\alpha=\beta=\gamma\ne0$.
\end{conds}
\begin{options}
\item Only (i), (iii), and (iv)
\item Only (iv)
\item Only (ii)
\item Only (i) and (iii)
\end{options}

\solution
By \eqref{eq:F-rank-pivots} the rank is the number of non-zero rows in the row echelon form.

Case (i), $\alpha=0$ and $\beta=\gamma\ne0$: rows 3 and 4 are equal, so $R_4\to R_4-R_3$ gives a zero row.
\begin{align}
\vec A&=\myvec{2&3&4&5\\0&6&7&8\\0&0&0&\beta\\0&0&0&\beta}\xrightarrow{R_4\to R_4-R_3}\myvec{2&3&4&5\\0&6&7&8\\0&0&0&\beta\\0&0&0&0}
\\
\rank\brak{\vec A}&=3
\end{align}
Case (ii), $\alpha=\beta=\gamma=0$: the last two rows are zero.
\begin{align}
\vec A&=\myvec{2&3&4&5\\0&6&7&8\\0&0&0&0\\0&0&0&0}
\\
\rank\brak{\vec A}&=2
\end{align}
Case (iii), $\beta=\gamma=0$ and $\alpha\ne0$: only the fourth row is zero.
\begin{align}
\vec A&=\myvec{2&3&4&5\\0&6&7&8\\0&0&\alpha&0\\0&0&0&0}
\\
\rank\brak{\vec A}&=3
\end{align}
Case (iv), $\alpha=\beta=\gamma\ne0$: the matrix is already in echelon form with no zero row.
\begin{align}
\vec A&=\myvec{2&3&4&5\\0&6&7&8\\0&0&\alpha&\alpha\\0&0&0&\alpha}
\\
\rank\brak{\vec A}&=4
\end{align}


\item\label{p:series}
Consider the following series:
\begin{align}
\text{(i)}\ \sum_{n=1}^{\infty}\frac{1}{\sqrt n},\qquad
\text{(ii)}\ \sum_{n=1}^{\infty}\frac{1}{n(n+1)},\qquad
\text{(iii)}\ \sum_{n=1}^{\infty}\frac{1}{n!}
\end{align}
Choose the correct option. \hfill (EC~2025)
\begin{options}
\item Only (ii) converges
\item Only (ii) and (iii) converge
\item Only (iii) converges
\item All three converge
\end{options}

\solution
Series (i) is a $p$-series, series (ii) telescopes and series (iii) passes the ratio test:
\begin{align}
\sum_{n=1}^{\infty}\frac{1}{\sqrt n}&=\sum_{n=1}^{\infty}n^{-1/2},\qquad p=\frac12\le1\ \Rightarrow\ \text{diverges}\uses{eq:F-pseries}
\\
\sum_{n=1}^{N}\frac{1}{n(n+1)}&=1-\frac{1}{N+1}\ \to\ 1\uses{eq:F-telescope}
\\
\frac{a_{n+1}}{a_n}&=\frac{n!}{\brak{n+1}!}=\frac{1}{n+1}\ \to\ 0<1\uses{eq:F-ratio}
\\
\sum_{n=1}^{\infty}\frac{1}{n!}&=e-1\uses{eq:F-exp}
\end{align}

\item\label{p:balls}
A pot contains two red balls and two blue balls. Two balls are drawn from this pot randomly without replacement. What is the probability that the two balls drawn have different colours? \hfill (EC~2025)
\begin{options}
\item $2/3$
\item $1/3$
\item $1/2$
\item $1$
\end{options}

\solution
Number the balls $1,2$ (red) and $3,4$ (blue). Choosing two balls at random is the same as selecting each ball independently with probability $\tfrac12$ and keeping the outcomes with exactly two balls selected. Let $B$ be the selection indicator of one ball, with $p_B[0]=p_B[1]=\tfrac12$. The number $X$ of selected red balls is the sum of two independent indicators, so by the convolution formula \eqref{eq:F-conv-pmf}:
\begin{align}
p_X[0]&=p_B[0]\,p_B[0]=\frac14
\\
p_X[1]&=p_B[0]\,p_B[1]+p_B[1]\,p_B[0]=\frac12
\\
p_X[2]&=p_B[1]\,p_B[1]=\frac14
\end{align}
The number $Y$ of selected blue balls has the same PMF, $p_Y=p_X$. The total number selected is $S=X+Y$, and \eqref{eq:F-conv-pmf} gives every case:
\begin{align}
p_S[0]&=p_X[0]\,p_Y[0]=\frac1{16}
\\
p_S[1]&=p_X[0]\,p_Y[1]+p_X[1]\,p_Y[0]=\frac14
\\
p_S[2]&=p_X[0]\,p_Y[2]+p_X[1]\,p_Y[1]+p_X[2]\,p_Y[0]=\frac38
\\
p_S[3]&=p_X[1]\,p_Y[2]+p_X[2]\,p_Y[1]=\frac14
\\
p_S[4]&=p_X[2]\,p_Y[2]=\frac1{16}
\end{align}
Exactly two balls are drawn when $S=2$. They have different colours when $X=1$ and $Y=1$, whose probability is, by independence \eqref{eq:F-indep},
\begin{align}
p_X[1]\,p_Y[1]&=\frac14
\\
\pr{\text{different colours}}&=\frac{p_X[1]\,p_Y[1]}{p_S[2]}=\frac{1/4}{3/8}=\frac23
\end{align}

\includegraphics[width=1.00\linewidth]{figs/Two_balls.png}
\caption{PMFs of $X$, $Y$ and $S=X+Y$ (convolution)}
\label{fig:c-balls}
\end{figure}

\item\label{p:lti}
Consider the discrete-time system in \figref{fig:p5} with input $x[n]$ and output $y[n]$. Here $h_1[n]$ and $h_2[n]$ are the impulse responses of LTI Subsystems 1 and 2, $\delta[n]$ is the unit impulse, and $b>0$. Assuming $h_2[n]\ne\delta[n]$, the overall system (dashed box) is \underline{\hspace{1.5cm}}. \hfill (EC~2025)
\begin{options}
\item linear and time invariant
\item linear and time variant
\item nonlinear and time invariant
\item nonlinear and time variant
\end{options}
\begin{figure}[H]
\centering
\includegraphics[width=0.85\linewidth]{figs/fig-lti-system.png}
\caption{Discrete-time system with subsystems $h_1[n]$, $h_2[n]$ and the added impulses}
\label{fig:p5}

From the figure, $b\delta[n]$ is added after Subsystem~1 and $-b\delta[n]$ is added after Subsystem~2. Using $\delta[n]*h_2[n]=h_2[n]$ \eqref{eq:F-conv},

\end{align}
These differ (the term $-b\delta[n]$ stays at $n=0$ while $\delta[n-k]$ moves, for a stable $h_2$), so \eqref{eq:F-ti} fails and the system is time variant.

\begin{options}
\item $\abs{F(\omega)}\le\displaystyle\int_{-\infty}^{\infty}\abs{f(t)}\dd t$
\item $\abs{F(\omega)}>\displaystyle\int_{-\infty}^{\infty}\abs{f(t)}\dd t$
\item $\abs{F(\omega)}\le\displaystyle\int_{-\infty}^{\infty}f(t)\dd t$
\item $\abs{F(\omega)}\ge\displaystyle\int_{-\infty}^{\infty}f(t)\dd t$
\end{options}

\solution
Test the options with $f(t)=e^{-\abs t}$, for which $\int\abs f\dd t=\int f\dd t=2$. By \eqref{eq:F-ft-pair},

\begin{align}
\abs{F(\omega)}&\le\int_{-\infty}^{\infty}\abs{f(t)}\abs{e^{-\jj\omega t}}\dd t
\\
&=\int_{-\infty}^{\infty}\abs{f(t)}\dd t\uses{eq:F-ft-bound}
\end{align}

\item\label{p:cubic}
Consider $f:\mathbb R\to\mathbb R$ defined by $f(x)=2x^3-3x^2-12x+1$. Which of the following statements is/are correct? \hfill (EC~2025)
\begin{options}
\item $f$ has no global maximizer
\item $f$ has no global minimizer
\item $x=-1$ is a local minimizer of $f$
\item $x=2$ is a local maximizer of $f$
\end{options}

\solution
The critical points follow from $f'(x)=0$, and the second derivative classifies them \eqref{eq:F-second}:
\begin{align}
f'(x)&=6x^2-6x-12=6\brak{x+1}\brak{x-2}
\\
f''(x)&=12x-6
\\
f''(-1)&=-18<0\quad\text{(local maximum, }f(-1)=8)
\\
f''(2)&=18>0\quad\text{(local minimum, }f(2)=-19)
\end{align}
So (C) and (D) are false. Since $2x^3$ dominates,
\begin{align}
\lim_{x\to\infty}f(x)&=+\infty
\\
\lim_{x\to-\infty}f(x)&=-\infty
\end{align}
and $f$ has neither a global maximizer nor a global minimizer.

\item $\displaystyle\oint_Cz^n\dd z=0$, where $n$ is an even integer
\item $\displaystyle\oint_C\cos z\dd z\ne0$
\item $\displaystyle\oint_C\sec z\dd z\ne0$
\end{options}

\solution
The functions $e^z$ and $\cos z$ are analytic everywhere, so both integrals vanish \eqref{eq:F-cauchy}:
\begin{align}
\oint_Ce^z\dd z&=0\uses{eq:F-cauchy}
\\
\oint_C\cos z\dd z&=0\uses{eq:F-cauchy}
\end{align}
(A) is true and (C) is false. For integer $n\ne-1$, in particular every even $n$,
\begin{align}
\oint_Cz^n\dd z&=0\uses{eq:F-zn}
\end{align}
so (B) is true. The poles of $\sec z$ are at $\pm\pi/2,\pm3\pi/2,\dots$ \eqref{eq:F-sec}, all outside $\abs z=1$, so
\begin{align}
\oint_C\sec z\dd z&=0\uses{eq:F-cauchy}
\end{align}
and (D) is false.

\item\label{p:state}
Consider a system with internal states $x_1(t),x_2(t),x_3(t)$ and input $u(t)$ governed by
\begin{align}
\frac{d}{dt}\myvec{x_1\\x_2\\x_3}=\myvec{2&0&0\\0&-2&0\\0&0&0}\myvec{x_1\\x_2\\x_3}+\myvec{1\\1\\1}u(t)
\end{align}
Which of the following statements is/are TRUE? \hfill (EC~2025)
\begin{options}
\item The signals $x_1,x_2,x_3$ are bounded for all bounded inputs.
\item There exists a bounded input such that at least one of $x_1,x_2,x_3$ is unbounded.
\item There exists a bounded input such that the signals $x_1,x_2,x_3$ are unbounded.
\item The signals $x_1,x_2,x_3$ are unbounded for all bounded inputs.
\end{options}

\solution
Since $\vec A$ is diagonal, the states decouple:
\begin{align}
\dot x_1&=2x_1+u
\\
\dot x_2&=-2x_2+u
\\
\dot x_3&=u
\end{align}
For $u\equiv1$ and zero initial state, $x_1$ (eigenvalue $+2$) and $x_3$ (eigenvalue $0$, an integrator) grow without bound, while $x_2$ (eigenvalue $-2$) stays bounded for every bounded input:

x_1(t)&=\frac12\brak{e^{2t}-1}\to\infty\uses{eq:F-state-unbounded}
\\
x_3(t)&=t\to\infty\uses{eq:F-state-integrator}
\\
\abs{x_2(t)}&\le\abs{x_2(0)}+\frac{U}{2}\uses{eq:F-state-bounded}
\end{align}
So (B) is true. (C) and (D) need all three signals unbounded, which fails because of $x_2$; (A) is false.

\item\label{p:entropy}
The random variable $X$ takes values in $\{-1,0,1\}$ with $\pr{X=-1}=\pr{X=1}=\alpha$ and $\pr{X=0}=1-2\alpha$, where $0<\alpha<\tfrac12$. Let $g(\alpha)$ be the entropy of $X$ in bits. Which of the following is/are TRUE? \hfill (EC~2025)
\begin{options}
\item $g(0.4)>g(0.3)$
\item $g(0.3)>g(0.4)$
\item $g(0.3)>g(0.25)$
\item $g(0.25)>g(0.3)$
\end{options}

\solution
\begin{align}
g(\alpha)&=-2\alpha\log_2\alpha-\brak{1-2\alpha}\log_2\brak{1-2\alpha}\uses{eq:F-entropy}
\\
g'(\alpha)&=2\log_2\frac{1-2\alpha}{\alpha}
\end{align}
$g'>0$ for $\alpha<\tfrac13$, so $g$ increases up to $\tfrac13$: $g(0.3)>g(0.25)$. The points $0.3$ and $0.4$ lie on opposite sides of the maximum, so evaluate directly:
\begin{align}
g(0.25)&=0.5\cdot2+0.5\cdot1=1.5000
\\
g(0.3)&=-0.6\log_20.3-0.4\log_20.4=1.5710
\\
g(0.4)&=-0.8\log_20.4-0.2\log_20.2=1.5219
\end{align}
Hence $g(0.3)>g(0.4)$ and $g(0.3)>g(0.25)$.

f(t)=\sum_{k=-\infty}^{\infty}c_k\,e^{\jj\frac{2\pi}{T_0}kt},\qquad
y(t)=\sum_{k=-\infty}^{\infty}d_k\,e^{\jj\frac{2\pi\alpha}{T_0}kt}
\end{align}
Which of the following is/are TRUE? \hfill (EC~2025)
\begin{options}

\item $y(t)$ is periodic with fundamental period $\alpha T_0$

\item\label{p:mi}
$X$ and $Y$ are Bernoulli random variables on $\{0,1\}$ with joint pmf
\begin{align}
\pr{0,0}=0.06,\quad\pr{0,1}=0.14,\quad\pr{1,0}=0.24,\quad\pr{1,1}=0.56
\end{align}

\solution
The marginals follow by summing the joint PMF \eqref{eq:F-marginal}:
\begin{align}
\pr{X=0}&=0.06+0.14=0.20
\\
\pr{X=1}&=0.24+0.56=0.80
\\
\pr{Y=0}&=0.06+0.24=0.30
\\

\\
I\brak{X;Y}&=0\uses{eq:F-mi-zero}
\end{align}
